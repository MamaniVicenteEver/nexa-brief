import asyncio,json,statistics,sys
import numpy as np
from pathlib import Path
from playwright.async_api import async_playwright
sys.path.insert(0,'/tmp');from cortes_matrix3 import TPL,FILES,BACK,URL
OUT=TPL/'work/results/cortes-visual';OUT.mkdir(exist_ok=True)
RANGES=[(3,10),(19.8,30),(51,55),(93,97)]
async def main():
 results=[]
 async with async_playwright() as p:
  b=await p.chromium.launch(args=['--autoplay-policy=no-user-gesture-required'])
  for orient,(w,h) in [('16x9',(1280,720)),('9x16',(720,1280))]:
   page=await b.new_page(viewport={'width':w,'height':h});await page.goto(URL+'?paused&t=0');await page.wait_for_function('window.__promo && window.__promo.beats',timeout=120000);await page.evaluate('window.KitReady')
   await page.evaluate("window.__small=document.createElement('canvas');__small.width=160;__small.height=90;window.__sg=__small.getContext('2d',{willReadFrequently:true})")
   for lo,hi in RANGES:
    rows=[];prev=None
    for i in range(round(lo*30),round(hi*30)+1):
     t=i/30;await page.evaluate('(t)=>__promo.seek(t)',t)
     rgb=await page.evaluate("() => {__sg.drawImage(document.querySelector('.k-main'),0,0,160,90);return Array.from(__sg.getImageData(0,0,160,90).data);}")
     a=np.asarray(rgb,dtype=np.int16).reshape(90,160,4)[:,:,:3]
     if prev is not None:
      delta=np.abs(a-prev);rows.append({'t':t,'mae':float(delta.mean()),'pixels_gt25':float(np.mean(delta.max(axis=2)>25))})
     prev=a
    median=statistics.median(r['mae'] for r in rows);peaks=sorted(rows,key=lambda r:-r['mae'])[:6]
    for r in peaks[:2]:
     for t in [r['t']-1/30,r['t']]:
      await page.evaluate('(t)=>__promo.seek(t)',t);await page.screenshot(path=str(OUT/f'{orient}-{t:.4f}.png'))
    result={'orient':orient,'range':[lo,hi],'median_mae':median,'peaks':peaks,'rows':rows};results.append(result);print(orient,lo,hi,'median',round(median,3),'peaks',peaks[:3],flush=True)
   await page.close()
  await b.close()
 (OUT/'summary.json').write_text(json.dumps(results,indent=2))
asyncio.run(main())
