# Executed only through Blender MCP, together with the model source.
def military_materials():
    land=not SPRITE.startswith(('ship-','air-'))
    olive=LIVERY=='olive' or SPRITE in ['air-transport-jet','air-helicopter-heavy']
    base=(.075,.105,.025,1) if olive else (.38,.265,.105,1) if LIVERY=='desert' else (.075,.105,.045,1) if land else (.105,.135,.155,1) if SPRITE.startswith('ship-') else (.065,.080,.090,1)
    for key,mat in AEM_M.items():
        ns=mat.node_tree.nodes; lk=mat.node_tree.links
        ns.clear(); out=ns.new('ShaderNodeOutputMaterial'); sh=ns.new('ShaderNodeBsdfPrincipled');lk.new(sh.outputs['BSDF'],out.inputs['Surface'])
        sh.inputs['Metallic'].default_value=.75 if key=='steel' else .25 if key=='seam' else 0
        sh.inputs['Roughness'].default_value=.42 if key=='steel' else .18 if key=='glass' else .96 if key in ['canvas','rubber','mud'] else .82
        special={'dark':(.012,.016,.011,1),'steel':(.09,.10,.075,1),'rubber':(.018,.020,.012,1),'glass':(.008,.018,.024,1),'seam':(.025,.030,.017,1),'red':(.55,.003,.004,1),'medical':(.82,.81,.75,1),'mud':(.048,.030,.012,1)}
        color=special.get(key,base)
        if key=='deck':color=tuple(c*.50 for c in base[:3])+(1,)
        if key=='panel':color=tuple(c*.80 for c in base[:3])+(1,)
        # White is only a local highlight now; medical panels have their own slot.
        if key=='white':color=tuple(c*.85 for c in base[:3])+(1,)
        sh.inputs['Base Color'].default_value=color; mat.diffuse_color=color
        if key in ['red','medical','glass','dark']:continue
        tc=ns.new('ShaderNodeTexCoord');tc.object=AEM['R']
        no=ns.new('ShaderNodeTexNoise');no.inputs['Scale'].default_value=1.15 if land else .13;no.inputs['Detail'].default_value=2.4;no.inputs['Roughness'].default_value=.65
        lk.new(tc.outputs['Object'],no.inputs['Vector'])
        ramp=ns.new('ShaderNodeValToRGB')
        ramp.color_ramp.elements[0].position=.2;ramp.color_ramp.elements[0].color=tuple(c*.72 for c in color[:3])+(1,)
        ramp.color_ramp.elements[1].position=.8;ramp.color_ramp.elements[1].color=tuple(c*1.18 for c in color[:3])+(1,)
        if LIVERY=='camo' and key in ['paint','light','panel','canvas','white']:
            ramp.color_ramp.interpolation='CONSTANT'
            ramp.color_ramp.elements[0].position=.34;ramp.color_ramp.elements[0].color=(.018,.026,.012,1)
            ramp.color_ramp.elements[1].position=.57;ramp.color_ramp.elements[1].color=(.23,.15,.055,1)
            e=ramp.color_ramp.elements.new(.44);e.color=(.075,.105,.032,1)
        lk.new(no.outputs['Fac'],ramp.inputs['Fac'])
        fine=ns.new('ShaderNodeTexNoise');fine.inputs['Scale'].default_value=28 if land else 4;fine.inputs['Detail'].default_value=2
        lk.new(tc.outputs['Object'],fine.inputs['Vector'])
        dust=ns.new('ShaderNodeValToRGB');dust.color_ramp.elements[0].position=.58;dust.color_ramp.elements[0].color=(0,0,0,1);dust.color_ramp.elements[1].position=.83;dust.color_ramp.elements[1].color=(.40,.40,.40,1)
        lk.new(fine.outputs['Fac'],dust.inputs['Fac'])
        mix=ns.new('ShaderNodeMixRGB');mix.inputs[2].default_value=(.19,.12,.044,1) if land else (.065,.055,.036,1)
        lk.new(dust.outputs['Color'],mix.inputs[0]);lk.new(ramp.outputs['Color'],mix.inputs[1])
        # Sparse, physically located scratches on bevels; no repeating UV tiles.
        geo=ns.new('ShaderNodeNewGeometry');edge=ns.new('ShaderNodeValToRGB');edge.color_ramp.elements[0].position=.51;edge.color_ramp.elements[0].color=(0,0,0,1);edge.color_ramp.elements[1].position=.61;edge.color_ramp.elements[1].color=(.45,.45,.45,1)
        lk.new(geo.outputs['Pointiness'],edge.inputs['Fac'])
        wear=ns.new('ShaderNodeMixRGB');wear.inputs[2].default_value=(.16,.13,.075,1);lk.new(edge.outputs['Color'],wear.inputs[0]);lk.new(mix.outputs['Color'],wear.inputs[1]);lk.new(wear.outputs['Color'],sh.inputs['Base Color'])
        bump=ns.new('ShaderNodeBump');bump.inputs['Strength'].default_value=.16;bump.inputs['Distance'].default_value=.009 if land else .035;lk.new(fine.outputs['Fac'],bump.inputs['Height']);lk.new(bump.outputs['Normal'],sh.inputs['Normal'])
