SPRITE="tank-light"
# Adaptación de originals/models.py.txt para MCP seguro; sin operadores de contexto.
# +Y = frente/proa; tamaños de construcción normalizados, metros reales en catálogo.
import bpy, bmesh, math, json
from mathutils import Vector, Matrix
AEM_M={}
AEM={}

def aem_mat(key,gray,metal=0,rough=.65,texture=True):
    mat=bpy.data.materials.new('AEM_'+SPRITE+'_'+key); mat.use_nodes=True
    color=(gray,gray,gray,1) if not isinstance(gray,tuple) else gray
    mat.diffuse_color=color
    sh=next(n for n in mat.node_tree.nodes if n.type=='BSDF_PRINCIPLED')
    sh.inputs['Base Color'].default_value=color; sh.inputs['Metallic'].default_value=metal; sh.inputs['Roughness'].default_value=rough
    if texture:
        ns=mat.node_tree.nodes; lk=mat.node_tree.links
        tc=ns.new('ShaderNodeTexCoord'); no=ns.new('ShaderNodeTexNoise'); no.inputs['Scale'].default_value=48; no.inputs['Detail'].default_value=2
        ra=ns.new('ShaderNodeValToRGB')
        ra.color_ramp.elements[0].color=tuple(c*.76 for c in color[:3])+(1,)
        ra.color_ramp.elements[1].color=tuple(min(c*1.14,1) for c in color[:3])+(1,)
        bu=ns.new('ShaderNodeBump'); bu.inputs['Strength'].default_value=.13; bu.inputs['Distance'].default_value=.012
        lk.new(tc.outputs['Generated'],no.inputs['Vector']); lk.new(no.outputs['Fac'],ra.inputs['Fac']); lk.new(ra.outputs['Color'],sh.inputs['Base Color']); lk.new(no.outputs['Fac'],bu.inputs['Height']); lk.new(bu.outputs['Normal'],sh.inputs['Normal'])
    AEM_M[key]=mat
    return mat

def aem_materials():
    for k,g,m,r in [('paint',.25,0,.68),('light',.46,0,.58),('panel',.32,0,.70),('dark',.022,0,.76),('steel',.32,.82,.38),('rubber',.018,0,.90),('glass',.035,.05,.13),('white',.83,0,.48),('canvas',.27,0,.95),('deck',.19,0,.82),('seam',.075,.4,.62)]:
        aem_mat(k,g,m,r,k not in ['glass','dark'])
    aem_mat('red',(.64,.006,.009,1),0,.55,False)

def aem_mesh(name,verts,faces,mat='paint',edge=0,smooth=False):
    me=bpy.data.meshes.new('AEM_'+SPRITE+'_'+name); me.from_pydata(verts,[],faces); me.update()
    bm=bmesh.new(); bm.from_mesh(me); bmesh.ops.recalc_face_normals(bm,faces=list(bm.faces)); bm.to_mesh(me); bm.free()
    ob=bpy.data.objects.new('AEM_'+SPRITE+'_'+name,me); AEM['C'].objects.link(ob); ob.parent=AEM['R']
    me.materials.append(AEM_M[mat])
    for f in me.polygons: f.use_smooth=smooth
    if edge:
        mod=ob.modifiers.new('Cantos físicos','BEVEL'); mod.width=edge; mod.segments=2
        mod=ob.modifiers.new('Normales ponderadas','WEIGHTED_NORMAL'); mod.keep_sharp=True
    return ob

def box(name,loc,size,mat='paint',edge=.04,angle=0):
    x,y,z=(s/2 for s in size); c=math.cos(angle); s=math.sin(angle)
    vs=[(loc[0]+c*a-s*b,loc[1]+s*a+c*b,loc[2]+d) for a,b,d in [(-x,-y,-z),(-x,-y,z),(-x,y,-z),(-x,y,z),(x,-y,-z),(x,-y,z),(x,y,-z),(x,y,z)]]
    return aem_mesh(name,vs,[(0,4,6,2),(1,3,7,5),(0,1,5,4),(2,6,7,3),(0,2,3,1),(4,5,7,6)],mat,edge)

def polygon(name,pts,z,h,mat='paint',edge=.04):
    n=len(pts); vs=[(x,y,z) for x,y in pts]+[(x,y,z+h) for x,y in pts]
    return aem_mesh(name,vs,[tuple(range(n-1,-1,-1)),tuple(range(n,n*2))]+[(i,(i+1)%n,(i+1)%n+n,i+n) for i in range(n)],mat,edge)

def rod(name,a,b,r=.10,mat='steel',n=16):
    a=Vector(a); b=Vector(b); d=b-a; q=d.to_track_quat('Z','Y'); v=[]
    for end in [a,b]:
        for i in range(n): v.append(tuple(end+q@Vector((r*math.cos(i*math.tau/n),r*math.sin(i*math.tau/n),0))))
    return aem_mesh(name,v,[tuple(range(n-1,-1,-1)),tuple(range(n,n*2))]+[(i,(i+1)%n,(i+1)%n+n,i+n) for i in range(n)],mat,0,True)

def sphere(name,loc,size,mat='paint'):
    bm=bmesh.new(); bmesh.ops.create_uvsphere(bm,u_segments=24,v_segments=12,radius=1)
    vs=[tuple(Vector(loc)+Vector((v.co.x*size[0],v.co.y*size[1],v.co.z*size[2]))) for v in bm.verts]
    bm.verts.index_update(); fs=[tuple(v.index for v in f.verts) for f in bm.faces]; bm.free()
    return aem_mesh(name,vs,fs,mat,0,True)

def loft(name,stations,mat='paint',sides=24):
    vs=[(x+w*math.cos(i*math.tau/sides),y,z+h*math.sin(i*math.tau/sides)) for y,w,h,z,x in stations for i in range(sides)]
    fs=[(j*sides+i,j*sides+(i+1)%sides,(j+1)*sides+(i+1)%sides,(j+1)*sides+i) for j in range(len(stations)-1) for i in range(sides)]
    fs += [tuple(range(sides-1,-1,-1)),tuple(range((len(stations)-1)*sides,len(stations)*sides))]
    return aem_mesh(name,vs,fs,mat,0,True)

def ring(name,x,y,z,r,width=.055,mat='steel'):
    n=40; verts=[]
    for zz in [z-width*.5,z+width*.5]:
        for rr in [r-width/2,r+width/2]:
            verts.extend((x+rr*math.cos(i*math.tau/n),y+rr*math.sin(i*math.tau/n),zz) for i in range(n))
    faces=[]
    for i in range(n):
        j=(i+1)%n
        faces.extend([(i,j,n+j,n+i),(2*n+i,3*n+i,3*n+j,2*n+j),(i,2*n+i,2*n+j,j),(n+i,n+j,3*n+j,3*n+i)])
    return aem_mesh(name,verts,faces,mat)

def line(name,pts,z,r=.04,mat='seam'):
    for a,b in zip(pts,pts[1:]): rod(name,(a[0],a[1],z),(b[0],b[1],z),r,mat,8)

    loft('Casco volumétrico',[(-le*.49,w*.27,1.6,0,0),(-le*.4,w*.48,2.5,.5,0),(le*.2,w*.49,2.5,.5,0),(le*.4,w*.32,2.1,.7,0),(le*.499,.07,.1,1.6,0)],'paint',32)
    polygon('Cintón oscuro',pts,1.8,1.25,'dark',.20)
    polygon('Cubierta',[(x*.96,y*.99) for x,y in pts],3.05,.35,'deck',.10)
    line('Regala',[(x*.965,y*.991) for x,y in pts]+[(pts[0][0]*.965,pts[0][1]*.991)],3.50,.07,'light')
    for sg in [-1,1]:
        for y in [-le*.44,-le*.36,le*.29,le*.35]:
            box('Bita',(sg*w*.31,y,3.8),(.40,.8,.7),'steel',.07)
        rod('Molinete',(sg*w*.10,le*.33,3.4),(sg*w*.10,le*.33,4),.55,'steel',24)
        line('Cadena ancla',[(sg*w*.10,le*.33),(sg*w*.21,le*.40)],3.55,.10,'dark')
    return 3.4

def bridge(x,y,w=7,le=12,z=3.4,h=5):
    polygon('Puente',[(x-w*.5,y-le*.5),(x-w*.5,y+le*.24),(x-w*.27,y+le*.5),(x+w*.27,y+le*.5),(x+w*.5,y+le*.24),(x+w*.5,y-le*.5)],z,h,'light',.25)
    box('Techo puente',(x,y,z+h+.2),(w*.95,le*.9,.4),'panel',.15)
    for a in [-.30,0,.30]: box('Ventana puente',(x+a*w,y+le*.31,z+h+.45),(w*.22,le*.13,.1),'glass',.04)
    for sg in [-1,1]:
        for a in [-.25,0,.25]: box('Ventana lateral',(x+sg*w*.43,y+a*le,z+h+.44),(w*.08,le*.14,.08),'glass',.02)
    return z+h+.5

def mast(x,y,z,h=8):
    rod('Mástil',(x,y,z),(x,y,z+h),.15,'steel',12)
    for dz,wi in [(h*.3,1.8),(h*.7,2.7)]: rod('Cruceta',(x-wi,y,z+dz),(x+wi,y,z+dz),.10,'light',10)
    box('Radar',(x,y,z+h-1),(3.2,.65,1.4),'panel',.10,.3)
    sphere('Radomo',(x,y,z+h+.5),(.7,.7,.65),'white')

def rail(x1,y1,x2,y2,z=4.2):
    rod('Pasamanos',(x1,y1,z),(x2,y2,z),.055,'light',8)
    for i in range(15):
        u=i/14; x=x1+(x2-x1)*u; y=y1+(y2-y1)*u
        rod('Puntal',(x,y,z-.7),(x,y,z),.045,'steel',8)

def aem_start(sprite):
    master=bpy.data.collections.get('AEM_vehiculos')
    if master is None:
        master=bpy.data.collections.new('AEM_vehiculos'); bpy.context.scene.collection.children.link(master)
    if bpy.data.collections.get('AEM_'+sprite): raise ValueError('Colección existente; no sobrescribir')
    AEM['C']=bpy.data.collections.new('AEM_'+sprite); master.children.link(AEM['C'])
    AEM['R']=bpy.data.objects.new('AEM_'+sprite+'_ROOT',None); AEM['C'].objects.link(AEM['R']); AEM['R'].location=(12000,0,0)
    aem_materials()
    if sprite.startswith('tank-') or sprite=='artillery-self-propelled': tank(sprite)
    elif sprite=='apc': apc()
    elif sprite=='artillery-towed': towed()
    elif sprite=='engineer': engineer()
    elif sprite.startswith('ship-'): ship(sprite)
    elif sprite.startswith('air-'): aircraft(sprite)
    else: truck(sprite)
    bpy.context.view_layer.update()
    print(json.dumps({'sprite':sprite,'objects':len(AEM['C'].objects),'root':AEM['R'].name,'location':list(AEM['R'].location)}))

aem_start(SPRITE)
