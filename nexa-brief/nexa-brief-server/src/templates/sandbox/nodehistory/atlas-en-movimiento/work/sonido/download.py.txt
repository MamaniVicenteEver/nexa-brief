"""Descarga pública reproducible: fichas CC0 + MP3; sin registrar biblioteca."""
import concurrent.futures,hashlib,json,subprocess,urllib.request
from pathlib import Path
ROOT=Path.cwd()
old=json.loads((ROOT/'work/research/sfxmint-fichas.json').read_text())['sounds']
def get(row):
 slug=row['id_requested']; url='https://sfxmint.com/api/v1/sounds/'+slug
 with urllib.request.urlopen(url,timeout=60) as r: meta=json.load(r)
 assert meta['license']=='CC0' and meta['slug']==slug
 url='https://sfxmint.com/dl/'+slug+'.mp3'
 with urllib.request.urlopen(url,timeout=90) as r:
  assert r.headers.get_content_type().startswith('audio/')
  raw=r.read(50*1024*1024)
 digest=hashlib.sha256(raw).hexdigest()
 assert digest==meta['format_metadata']['mp3']['sha256'],slug
 path=ROOT/'assets/sfx'/f'{slug}.mp3';path.write_bytes(raw)
 pcm=subprocess.check_output(['ffmpeg','-v','error','-threads','1','-i',str(path),'-ar','48000','-ac','2','-f','f32le','-'])
 duration=len(pcm)/8/48000
 print(slug,duration,flush=True)
 return slug,dict(meta,decoded_duration=duration,local_sha256=digest)
