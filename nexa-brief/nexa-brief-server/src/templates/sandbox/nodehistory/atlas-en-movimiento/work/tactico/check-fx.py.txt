FLAGS = ['--use-gl=angle', '--use-angle=swiftshader', '--enable-unsafe-swiftshader', '--ignore-gpu-blocklist', '--num-raster-threads=1', '--renderer-process-limit=1', '--disable-dev-shm-usage']
def deadline(*args):
    raise TimeoutError('Operación Chromium sin respuesta en 60 s')

class Handler(SimpleHTTPRequestHandler):
    def translate_path(self, path):
        # The copied workspace lacks this shared font. FX use no text; only
        # the unrelated labels module's top-level load needs a local substitute.
        if urlparse(path).path == '/_tools/explainer-kit/vendor/fonts/Archivo-900.ttf':
            return str(PROJECT/'assets/fonts/AtlasEditorial.ttf')
        return super().translate_path(path)
    def log_message(self, *args):
        pass

def sheet(rows, output):
    if not rows:
        return
    width, height, label = 180, 320, 24
    image = Image.new('RGB', (width * 6, (height + label) * len(rows)), '#081e2c')
    draw = ImageDraw.Draw(image)
    for row, shots in enumerate(rows):
        for column, (name, shot) in enumerate(shots):
            x, y = column * width, row * (height + label)
            draw.text((x + 3, y + 5), name, fill='#e9e3ba')
            with Image.open(shot) as frame:
                image.paste(frame.convert('RGB').resize((width, height), Image.Resampling.LANCZOS), (x, y + label))
    image.save(output, quality=93)
