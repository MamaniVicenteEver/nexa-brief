class Handler(SimpleHTTPRequestHandler):
    def translate_path(self,path):
        original=super().translate_path(path)
        # The isolated template fixture lacks the vendor font that labels.js
        # imports. Resolve that existing shared dependency read-only, never
        # change a shared file or substitute our new component during QA.
        if not Path(original).exists() and unquote(urlparse(path).path)=='/_tools/explainer-kit/vendor/fonts/Archivo-900.ttf':
            fallback=SOURCE/'_tools/explainer-kit/vendor/fonts/Archivo-900.ttf'
            if fallback.exists():
                report['readonlyFallbacks'].append(str(fallback));return str(fallback)
        return original
    def log_message(self,fmt,*args):
        if len(args)>1 and str(args[1]).isdigit() and int(args[1])>=400:report['httpErrors'].append(str(args))

def sheet(rows,path):
    widths=[270 if orient=='9x16' else 480 for _,orient,_ in rows]
    heights=[480 if orient=='9x16' else 270 for _,orient,_ in rows]
    canvas=Image.new('RGB',(1440,sum(h+32 for h in heights)),'#081e2c');d=ImageDraw.Draw(canvas);y=0
    for (label,orient,shots),w,h in zip(rows,widths,heights):
        for n,(shot_path,t) in enumerate(shots):
            x=n*480+(480-w)//2;d.text((n*480+10,y+8),f'{label} / {orient} / {t:.2f}s',fill='#f6edb4')
            with Image.open(shot_path) as im:canvas.paste(im.convert('RGB').resize((w,h),Image.Resampling.LANCZOS),(x,y+32))
        y+=h+32
    canvas.save(path,quality=94)
